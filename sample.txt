struct Counter;

enum Message { Increment, Decrement };

impl Component for Counter {

    type State = i32;
    type Message = Message;

    fn message(&self, message: Message, ctx: &mut Ctx<i32>) {
        match message {
            Message::Increment => *ctx.state_mut() += 1,
            Message::Decrement => *ctx.state_mut() -= 1,
        }
    }

    fn render(&self, ui: &mut Ui, ctx: &Ctx<i32>) {
        let title = fmt!("Count: {}", ctx.state());
        col(ui); 
           text(title, ui); 
           row(ui);
               text_button("+", Message::Increment, ui);
               text_button("-", Message::Decrement, ui);
           end(ui);
        end(ui);
    }
}
